```python
# server/app/secretary/schema_map.py
from __future__ import annotations
from psycopg.rows import dict_row
from app.secretary import pgdb

GROUPS = ('kb', 'cfg', 'qa', 'base')

def _q(sql, params=None, fetch=True):
    import psycopg
    with psycopg.connect(pgdb.DSN, autocommit=True, row_factory=dict_row) as conn:
        with conn.cursor() as cur:
            cur.execute(sql, params or ())
            return cur.fetchall() if fetch and cur.description else []

def ensure():
    _q("""create table if not exists cfg.page_entry (
              table_name varchar(120) primary key,
              page varchar(80) not null default '',
              updated_at timestamptz not null default now())""")

def _tables():
    return _q("""select table_schema s, table_name t, obj_description((table_schema||'.'||table_name)::regclass, 'pg_class') c
                   from information_schema.tables
                  where table_schema = any(%s) and table_type='BASE TABLE'
                  order by table_schema, table_name""", (list(GROUPS),))

def _cols(s):
    return _q("""select table_name t, column_name c, data_type d, is_nullable n, column_default df
                   from information_schema.columns where table_schema=%s""", (s,))

def overview():
    ensure()
    cols = {s: _cols(s) for s in GROUPS}
    pages = {r['table_name']: r['page'] for r in _q("select * from cfg.page_entry")}
    out = []
    for r in _tables():
        full = '%s.%s' % (r['s'], r['t'])
        n = _q('select count(*) c from %s' % full)[0]['c']
        out.append({'group': r['s'], 'table': r['t'], 'zh': r['c'] or '',
                    'purpose': '已登记' if r['c'] else '', 'cols': len(cols.get(r['s'], [])),
                    'rows': int(n), 'page': pages.get(full, ''), 'refs': []})
    return out

def table_detail(full):
    s, t = full.split('.', 1)
    cols = _q("""select c.column_name c, c.data_type d, c.is_nullable n, c.column_default df,
                        col_description((c.table_schema||'.'||c.table_name)::regclass, c.ordinal_position) cm
                   from information_schema.columns c
                  where c.table_schema=%s and c.table_name=%s order by c.ordinal_position""", (s, t))
    rows = _q('select * from %s.%s limit 20' % (s, t))
    return {'columns': [{'name': c['c'], 'type': c['d'], 'nullable': c['n'] == 'YES',
                         'default': c['df'], 'comment': c['cm'] or ''} for c in cols], 'rows': rows}

def set_comment(full, column, text):
    s, t = full.split('.', 1)
    _q('comment on column %s.%s.%s is %s', (s, t, column, text), fetch=False)
    return {'ok': True}
```

```tsx
// web/src/pages/SchemaMap.tsx
import { useEffect, useState } from 'react'
import { Card, Collapse, Table, Tag, Input, message } from 'antd'
import api from '../api'

type Row = { group: string; table: string; zh: string; purpose: string; cols: number
             rows: number; page: string; refs: string[] }

export default function SchemaMap() {
  const [rows, setRows] = useState<Row[]>([])
  const [detail, setDetail] = useState<Record<string, any>>({})

  useEffect(() => { api.get('/api/schema/overview').then(r => setRows(r.data.items)) }, [])
  const detailOf = (full: string) =>
    api.get(`/api/schema/table/${full}`).then(r =>
      setDetail(d => ({ ...d, [full]: r.data })))

  const groups = ['kb', 'cfg', 'qa', 'base']
  return (
    <Card title="库结构（information_schema 实时读取）">
      {groups.map(g => (
        <Card key={g} type="inner" title={`${g}（${rows.filter(r => r.group === g).length} 张表）`}>
          <Table size="small" rowKey={r => r.group + '.' + r.table}
            dataSource={rows.filter(r => r.group === g)}
            onRow={r => ({ onClick: () => detailOf(r.group + '.' + r.table) })}
            columns={[
              { title: '表', render: (_, r) => <code>{r.group}.{r.table}</code> },
              { title: '中文名', dataIndex: 'zh' },
              { title: '用途', dataIndex: 'purpose' },
              { title: '字段/行数', render: (_, r) => `${r.cols} / ${r.rows}` },
              { title: '页面入口', render: (_, r) => r.page
                  ? <Tag color="green">{r.page}</Tag>
                  : <Tag color="red">无入口</Tag> },
            ]} />
        </Card>
      ))}
      {Object.entries(detail).map(([full, d]: any) => (
        <Card key={full} type="inner" title={`字段：${full}`}>
          <Table size="small" rowKey="name" dataSource={d.columns} pagination={false}
            columns={[
              { title: '字段', dataIndex: 'name' },
              { title: '类型', dataIndex: 'type' },
              { title: '可空', render: (_, r) => r.nullable ? '是' : '否' },
              { title: '含义', render: (_, r) => r.comment || <Tag>未登记</Tag> },
            ]} />
        </Card>
      ))}
    </Card>
  )
}
```
