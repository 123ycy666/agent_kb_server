# 每日预警推送 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增 `POST /api/alerts/daily`：上游传登录账号，返回该账号今天该办的一件事一条预警（一句话），只返回有异常的。

**Architecture:** 三层，各自一个文件。`alert_scope.py` 把账号解析成「角色 + 所 id 列表」（4 张表，失败关闸）；`alerts.py` 用一套 SQL 算 10 类判据并按角色过滤、生成一句话；`server.py` 只加路由。判据口径与一句话模板都放配置，不写死在逻辑里。

**Tech Stack:** Python 3.12 / pymysql（只读，复用 `tools_db._exec`）/ 现有 `server.py` 的 `BaseHTTPRequestHandler` 路由。

## Global Constraints

- **只读**：接口只 SELECT，绝不写业务库（`dlj_data`）。落库/推送由上游做。
- **不接受上游传 `role` / `scope`**：角色与范围一律现查，传了忽略。
- **解析失败一律关闸**：返回 `status="unknown"` + `alerts=[]`，**绝不默认全市**。
- **不写死业务数**：代码里不出现 `1910`、`47`、`112.11` 这类业务数字。
- **一条一件事**：预算花不完、审批积压是两件事 → 两条。
- **一句话**：`title` ≤ 40 汉字，范围 + 结论 + 最多 2 个数。
- **语气**：用「未审批 / 未审核 / 待审批 / 已通过」；**禁止**「卡在 / 单子 / 卡住 / 超标」。
- **花得完不发**：预算预警是预测（`年底预计 < 分配额 × 0.6`），不触发的所不返回。
- 所有金额单位：接口内一律**元**（原值），展示折万元由调用方定。

```python
def elapsed_months(today=None):
    """年初到今天过了几个月（按 30.4 天/月）。用于把当前速度外推到全年。"""
    t = today or time.strftime('%Y-%m-%d')
    try:
        y, m, d = [int(x) for x in str(t)[:10].split('-')]
    except Exception:                                   # noqa: BLE001
        y, m, d = time.localtime()[:3]
    doy = 0
    for mm in range(1, m):
        doy += [31, 29 if (y % 4 == 0 and (y % 100 != 0 or y % 400 == 0)) else 28,
                31, 30, 31, 30, 31, 31, 30, 31, 30, 31][mm - 1]
    doy += d
    return max(1.0, doy / 30.4)
```

```python
# -*- coding: utf-8 -*-
"""账号 → 角色 + 数据范围（预警鉴权的唯一入口）。

**为什么不用 run_sql**：run_sql 有业务字典白名单，且我们这里要拿的是「所 id 列表」，
是鉴权数据不是业务取数。所以直连 tools_db._exec（同样只读、同样有只读闸）。

**失败一律关闸**：查不到 / 停用 / 没角色 / 映射不到所 → ok=False，**绝不默认全市**。
"""
import alert_rules
import tools_db

SCHEMA = tools_db.config.SCHEMA


def _rows(sql, args=None, limit=200):
    r = tools_db._exec(sql, args, limit_rows=limit)
    return [{(k or '').lower(): v for k, v in row.items()} for row in (r.get('rows') or [])]


def _ids(rows, key):
    return [int(r[key]) for r in rows if r.get(key) is not None]


def _stations_by_dept(dept_ids):
    """部门 id 列表 → 供电所（t_power_company）。**按 department_id 对，不按名字对**。"""
    if not dept_ids:
        return []
    marks = ','.join(['%s'] * len(dept_ids))
    return _rows('SELECT id, name, department_id FROM t_power_company '
                 'WHERE deleted_flag=0 AND department_id IN (' + marks + ') '
                 'ORDER BY id', list(dept_ids))


def _dept_tree(dept_id):
    """该部门 + 其全部子部门 id。"""
    out = [int(dept_id)]
    rows = _rows('SELECT department_id, parent_id FROM t_department')
    changed = True
    while changed:
        changed = False
        for r in rows:
            d, p = r.get('department_id'), r.get('parent_id')
            if d is None or p is None:
                continue
            if int(p) in out and int(d) not in out:
                out.append(int(d))
                changed = True
    return out


def _dept(dept_id):
    rows = _rows('SELECT department_id, department_name, parent_id FROM t_department '
                 'WHERE department_id=%s', [dept_id], limit=1)
    return rows[0] if rows else {}


def _county_scope(st) -> str:
    """范围名与类型：全市 / 县公司 / 单个所。"""
    n = len(st['station_ids'])
    if n == 0:
        return {'scope': '', 'scope_type': 'unknown'}
    if n >= 47:
        return {'scope': '全市', 'scope_type': 'city'}
    if n > 1:
        return {'scope': st['station_names'][0] and st['dept_name'], 'scope_type': 'county'}
    return {'scope': st['station_names'][0], 'scope_type': 'station'}
```

**注意**：上面 `_county_scope` 的 `n >= 47` 是**判据，不是业务数**——「全市 = 全部所」用「部门是市公司」判断更准。改成：

```python
def _scope_of(dept_id, dept, stations):
    if int(dept_id) == 1:
        return {'scope': '全市', 'scope_type': 'city'}
    n = len(stations)
    if n == 1:
        return {'scope': stations[0]['name'], 'scope_type': 'station'}
    return {'scope': dept.get('department_name') or '', 'scope_type': 'county'}
```

```python
def resolve(account):
    acct = (account or '').strip()
    if not acct:
        return {'ok': False, 'account': acct, 'error': 'empty_account'}
    emp = _rows('SELECT employee_id, login_name, actual_name, department_id, '
                'disabled_flag, deleted_flag, administrator_flag '
                'FROM t_employee WHERE login_name=%s LIMIT 1', [acct], limit=1)
    if not emp:
        return {'ok': False, 'account': acct, 'error': 'no_such_account'}
    e = emp[0]
    if int(e.get('deleted_flag') or 0) or int(e.get('disabled_flag') or 0):
        return {'ok': False, 'account': acct, 'error': 'disabled'}

    roles = _rows('SELECT r.role_code, r.role_name FROM t_role_employee re '
                  'JOIN t_role r ON r.role_id=re.role_id WHERE re.employee_id=%s',
                  [e['employee_id']])
    codes = [str(r['role_code']) for r in roles if r.get('role_code')]
    if not codes:
        return {'ok': False, 'account': acct, 'error': 'no_role'}
    # 一个账号挂多个角色时取权限最大的（预警条数由角色归属表决定）
    prio = ['master', 'sgscwb', 'xgscwb', 'cto', 'yxb', 'sz', 'codea']
    role_code = next((c for c in prio if c in codes), codes[0])

    dept_id = int(e['department_id'])
    dept = _dept(dept_id)
    stations = _stations_by_dept(_dept_tree(dept_id))

    # 例外：县级角色（cto/yxb/xgscwb）挂在所级部门时，上溯父部门取范围
    # 实例：广水的 xgscwb003/xgsyjb003/xgsyxb003 部门挂在关庙供电所下
    if not stations and role_code in ('cto', 'yxb', 'xgscwb'):
        pid = dept.get('parent_id')
        if pid and int(pid) != 0:
            parent = _dept(pid)
            stations = _stations_by_dept(_dept_tree(pid))
            if stations:
                dept_id, dept = int(pid), parent

    if not stations:
        return {'ok': False, 'account': acct, 'error': 'no_station'}

    sc = _scope_of(dept_id, dept, stations)
    node = ''
    for n, c in alert_rules.LEDGER_ROLE_BY_NODE.items():
        if c == role_code:
            node = n
            break
    return {
        'ok': True, 'account': acct, 'employee_id': int(e['employee_id']),
        'name': e.get('actual_name') or '', 'role_code': role_code,
        'role_name': alert_rules.ROLE_NAMES.get(role_code, role_code),
        'dept_id': dept_id, 'dept_name': dept.get('department_name') or '',
        'parent_dept_id': dept.get('parent_id'),
        'scope': sc['scope'], 'scope_type': sc['scope_type'],
        'station_ids': [int(s['id']) for s in stations],
        'station_names': [s['name'] for s in stations],
        'ledger_role': role_code, 'ledger_node': node,
    }
```
